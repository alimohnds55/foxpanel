const express = require('express');
const { spawn } = require('child_process');
const crypto = require('crypto');
const fs = require('fs');

const app = express();
const PORT = process.env.PORT || 8080;

// متغیرهای قابل تنظیم در بخش Variables سایت Cloady
const PANEL_PASS = process.env.PANEL_PASS || 'admin123';
const USER_UUID = process.env.UUID || crypto.randomUUID();
const WS_PATH = process.env.WS_PATH || '/ws-tunnel';

// ساخت فایل کانفیگ Xray
const xrayConfig = {
  inbounds: [{
    port: 10000,
    listen: "127.0.0.1",
    protocol: "vless",
    settings: {
      clients: [{ id: USER_UUID }],
      decryption: "none"
    },
    streamSettings: {
      network: "ws",
      wsSettings: { path: WS_PATH }
    }
  }],
  outbounds: [{ protocol: "freedom" }]
};

fs.writeFileSync('/app/config.json', JSON.stringify(xrayConfig));

// اجرای هسته Xray در پس‌زمینه
spawn('/app/xray', ['run', '-c', '/app/config.json'], { stdio: 'inherit' });

const httpProxy = require('http');

// هدایت درخواست‌های WebSocket وی‌پی‌ان به هسته Xray
const net = require('net');
server.on('upgrade', (req, socket, head) => {
  if (req.url.startsWith(WS_PATH)) {
    const proxySocket = net.connect(10000, '127.0.0.1', () => {
      proxySocket.write(
        `${req.method} ${req.url} HTTP/${req.httpVersion}\r\n` +
        Object.keys(req.headers).map(k => `${k}: ${req.headers[k]}`).join('\r\n') +
        '\r\n\r\n'
      );
      proxySocket.write(head);
      socket.pipe(proxySocket).pipe(socket);
    });
    proxySocket.on('error', () => socket.end());
  } else {
    socket.destroy();
  }
});

server.listen(PORT, () => console.log('Panel & Xray running on port ' + PORT));
